> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add keyboard lock mode with 5-minute idle auto-lock, 4-corner-hold unlock/lock combo with progress bar, and denied-keypress red flash.

**Architecture:** State machine driven by `matrix_scan_user` (idle timer + combo detection) and `process_record_user` (key blocking). All visuals in `rgb_matrix_indicators_advanced_user`. Corner key positions and number-row LED indices discovered at boot via keymap scan, so no hardcoded matrix coordinates.

**Tech Stack:** QMK firmware (C), RGB Matrix API, RP2040 target

```c
static void lock_find_positions(void) {
    // Corner keys: GRV (top-left), EQL (top-right), Z (bottom-left), SLSH (bottom-right)
    const uint16_t corner_keys[4] = {KC_GRV, KC_EQL, KC_Z, KC_SLSH};
    // Number row keys in visual left-to-right order
    const uint16_t numrow_keys[10] = {
        KC_1, KC_2, KC_3, KC_4, KC_5, KC_6, KC_7, KC_8, KC_9, KC_0
    };

    for (uint8_t r = 0; r < MATRIX_ROWS; r++) {
        for (uint8_t c = 0; c < MATRIX_COLS; c++) {
            uint16_t kc = keymap_key_to_keycode(_BASE, (keypos_t){.row = r, .col = c});
            for (uint8_t i = 0; i < 4; i++) {
                if (kc == corner_keys[i]) {
                    corner_pos[i] = (keypos_t){.row = r, .col = c};
                }
            }
            for (uint8_t i = 0; i < 10; i++) {
                if (kc == numrow_keys[i]) {
                    numrow_leds[i] = g_led_config.matrix_co[r][c];
                }
            }
        }
    }
}
```

### Task 3: Add matrix_scan_user with idle timer and combo detection

**Files:**
- Modify: `firmware/keyboards/splitkb/halcyon/elora/keymaps/denis/keymap.c`

**Step 1: Add `lock_engage` helper and `matrix_scan_user` after `keyboard_post_init_user`**

```c
// ── Lock mode helpers ──

static void lock_engage(void) {
    keyboard_locked = true;
    layer_state_set(1 << _BASE);  // force base layer only
    clear_oneshot_mods();
    lock_combo_start = 0;
}

static void lock_disengage(void) {
    keyboard_locked = false;
    lock_combo_start = 0;
}

// ── Matrix scan: idle lock + combo detection ──

void matrix_scan_user(void) {
    // Auto-lock after idle timeout
    if (!keyboard_locked && last_input_activity_elapsed() > LOCK_IDLE_TIMEOUT) {
        lock_engage();
    }

    // Check if all 4 corner keys are physically held
    bool all_corners = true;
    for (uint8_t i = 0; i < 4; i++) {
        if (!matrix_is_on(corner_pos[i].row, corner_pos[i].col)) {
            all_corners = false;
            break;
        }
    }

    if (all_corners) {
        if (lock_combo_start == 0) {
            lock_combo_start = timer_read32();
        } else if (timer_elapsed32(lock_combo_start) >= LOCK_COMBO_HOLD_MS) {
            if (keyboard_locked) {
                lock_disengage();
            } else {
                lock_engage();
            }
        }
    } else {
        lock_combo_start = 0;
    }
}
```

### Task 4: Add key blocking in process_record_user

**Files:**
- Modify: `firmware/keyboards/splitkb/halcyon/elora/keymaps/denis/keymap.c`

**Step 1: Add lock check at the top of the existing `process_record_user` function**

The current function starts with the RGB keycode switch. Insert the lock check before it:

```c
bool process_record_user(uint16_t keycode, keyrecord_t *record) {
    // Lock mode: block all keys and flash denied LED
    if (keyboard_locked && lock_combo_start == 0) {
        if (record->event.pressed) {
            uint8_t r = record->event.key.row;
            uint8_t c = record->event.key.col;
            uint8_t led = g_led_config.matrix_co[r][c];
            if (led != NO_LED) {
                denied_key_led     = led;
                denied_flash_start = timer_read32();
            }
        }
        return false;
    }

    // Existing RGB keycode blocking...
    switch (keycode) {
        case RM_TOGG: case RM_NEXT: case RM_PREV:
        // ... rest unchanged
    }
}
```

Note: the `lock_combo_start == 0` condition allows keypresses through during the combo hold so the combo detection (which uses `matrix_is_on`, not keycodes) isn't affected. Actually, we still block them — the combo detection is in `matrix_scan_user` which reads physical matrix state regardless. But by checking `lock_combo_start == 0`, we avoid recording denied flashes while the user is mid-combo.

```c
bool rgb_matrix_indicators_advanced_user(uint8_t led_min, uint8_t led_max) {
    if (!rgb_matrix_is_enabled()) {
        rgb_matrix_enable_noeeprom();
    }

    // ── Lock mode visuals ──
    if (keyboard_locked || lock_combo_start != 0) {
        // Start with all LEDs off
        for (uint8_t i = led_min; i < led_max; i++) {
            rgb_matrix_set_color(i, 0, 0, 0);
        }

        // Progress bar on number row during combo hold
        if (lock_combo_start != 0) {
            uint32_t elapsed = timer_elapsed32(lock_combo_start);
            uint8_t lit_count = elapsed / (LOCK_COMBO_HOLD_MS / LOCK_PROGRESS_STEPS);
            if (lit_count > LOCK_PROGRESS_STEPS) lit_count = LOCK_PROGRESS_STEPS;

            for (uint8_t i = 0; i < lit_count; i++) {
                if (numrow_leds[i] >= led_min && numrow_leds[i] < led_max) {
                    rgb_matrix_set_color(numrow_leds[i], 0, 200, 0);  // green
                }
            }

            // Dim green underglow pulse during combo
            for (uint8_t i = led_min; i < led_max; i++) {
                if (HAS_FLAGS(g_led_config.flags[i], LED_FLAG_UNDERGLOW)) {
                    uint8_t brightness = 40 + 40 * ((timer_read() % 1000) < 500);
                    rgb_matrix_set_color(i, 0, brightness, 0);
                }
            }
            return false;
        }

        // Denied keypress: red flash on the specific key
        if (denied_key_led != NO_LED && timer_elapsed32(denied_flash_start) < LOCK_DENIED_FLASH_MS) {
            if (denied_key_led >= led_min && denied_key_led < led_max) {
                rgb_matrix_set_color(denied_key_led, 255, 0, 0);
            }
        } else {
            denied_key_led = NO_LED;
        }

        return false;
    }

    // ── Normal mode visuals (existing code below) ──
    uint8_t layer = get_highest_layer(layer_state | default_layer_state);
```
